cmake_minimum_required(VERSION 3.21)
project(RefreshComparison VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Qml Quick QuickControls2)

add_library(refresh_common STATIC common/RefreshController.cpp common/RefreshController.h)
target_link_libraries(refresh_common PUBLIC Qt6::Core Qt6::Gui)
target_include_directories(refresh_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/common)
add_executable(widgets_refresh widgets/main.cpp)
target_link_libraries(widgets_refresh PRIVATE refresh_common Qt6::Widgets)
add_executable(quick_refresh quick/main.cpp)
target_link_libraries(quick_refresh PRIVATE refresh_common Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_resources(quick_refresh "qml" PREFIX "/" FILES quick/Main.qml)
if(MSVC)
    target_compile_options(refresh_common PRIVATE /utf-8)
    target_compile_options(widgets_refresh PRIVATE /utf-8)
    target_compile_options(quick_refresh PRIVATE /utf-8)
endif()
